cmake_minimum_required(VERSION 3.22)
project(vosk_stt_windows LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(GODOT_CPP_DIR "" CACHE PATH "Path to a matching godot-cpp checkout")
set(VOSK_SDK_DIR "" CACHE PATH "Path to the Vosk native SDK")
set(GDEXTENSION_OUTPUT_DIR "${CMAKE_CURRENT_LIST_DIR}/bin" CACHE PATH
	"Directory for the DLL consumed by vosk_stt.gdextension")

if(NOT WIN32)
	message(FATAL_ERROR "The Vosk GDExtension currently supports Windows only.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
	message(FATAL_ERROR "The Vosk GDExtension requires a 64-bit Windows toolchain.")
endif()
if(NOT EXISTS "${GODOT_CPP_DIR}/CMakeLists.txt")
	message(FATAL_ERROR "GODOT_CPP_DIR must point to a matching godot-cpp checkout.")
endif()
if(NOT EXISTS "${VOSK_SDK_DIR}/include/vosk_api.h")
	message(FATAL_ERROR "VOSK_SDK_DIR must contain the Vosk SDK headers.")
endif()

add_subdirectory("${GODOT_CPP_DIR}" godot-cpp)
add_library(vosk_stt SHARED vosk_stt_windows.cpp)
target_include_directories(vosk_stt PRIVATE "${VOSK_SDK_DIR}/include")
target_link_directories(vosk_stt PRIVATE "${VOSK_SDK_DIR}/lib")
target_link_libraries(vosk_stt PRIVATE godot-cpp vosk)
target_compile_definitions(vosk_stt PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)

set_target_properties(vosk_stt PROPERTIES
	OUTPUT_NAME "vosk_stt.windows.x86_64"
	RUNTIME_OUTPUT_DIRECTORY "${GDEXTENSION_OUTPUT_DIR}"
	LIBRARY_OUTPUT_DIRECTORY "${GDEXTENSION_OUTPUT_DIR}"
	ARCHIVE_OUTPUT_DIRECTORY "${GDEXTENSION_OUTPUT_DIR}"
	RUNTIME_OUTPUT_DIRECTORY_DEBUG "${GDEXTENSION_OUTPUT_DIR}"
	RUNTIME_OUTPUT_DIRECTORY_RELEASE "${GDEXTENSION_OUTPUT_DIR}"
	RUNTIME_OUTPUT_DIRECTORY_RELWITHDEBINFO "${GDEXTENSION_OUTPUT_DIR}"
	RUNTIME_OUTPUT_DIRECTORY_MINSIZEREL "${GDEXTENSION_OUTPUT_DIR}"
	LIBRARY_OUTPUT_DIRECTORY_DEBUG "${GDEXTENSION_OUTPUT_DIR}"
	LIBRARY_OUTPUT_DIRECTORY_RELEASE "${GDEXTENSION_OUTPUT_DIR}"
	LIBRARY_OUTPUT_DIRECTORY_RELWITHDEBINFO "${GDEXTENSION_OUTPUT_DIR}"
	LIBRARY_OUTPUT_DIRECTORY_MINSIZEREL "${GDEXTENSION_OUTPUT_DIR}"
	ARCHIVE_OUTPUT_DIRECTORY_DEBUG "${GDEXTENSION_OUTPUT_DIR}"
	ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${GDEXTENSION_OUTPUT_DIR}"
	ARCHIVE_OUTPUT_DIRECTORY_RELWITHDEBINFO "${GDEXTENSION_OUTPUT_DIR}"
	ARCHIVE_OUTPUT_DIRECTORY_MINSIZEREL "${GDEXTENSION_OUTPUT_DIR}"
)
